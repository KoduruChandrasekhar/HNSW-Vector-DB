function(vdb_test name label)
  add_executable(${name} ${name}.cc)
  target_link_libraries(${name} PRIVATE vdb_internal GTest::gtest_main vdb_warnings)
  add_test(NAME ${name} COMMAND ${name})
  set_tests_properties(${name} PROPERTIES LABELS "${label}" TIMEOUT 900)
endfunction()
